/* ===================================== CALCULAR DISTÂNCIA ===================================== */ if ( percurso.length > 1 ) { const anterior = percurso[ percurso.length - 2 ]; distanciaHistorica += calcularDistancia( anterior[0], anterior[1], lat, lng ); /* ===================================== DESENHAR LINHA ===================================== */ linhaPercurso .setLatLngs( percurso ); /* ===================================== HISTÓRICO ===================================== */ if (!aoVivo) { const select = document.getElementById( "listaPercursos" ); } else { texto += "📚 PERCURSO SELECIONADO"; } /* ===================================== DATA ===================================== */ if (primeiroTimestamp) { const inicio = new Date( primeiroTimestamp ); texto += "

🕐 Início: " + inicio.toLocaleTimeString( "pt-BR" ) + ""; } /* ===================================== FIM ===================================== */ if (ultimoTimestamp) { const fim = new Date( ultimoTimestamp ); texto += "
